  var valueObj : object = clientObj.MakeValue();

  {
    raise "IValue(valueObj) should be a wrapper object in clientMain";
  }
  var value = cast (RpcDtorTest::IValue^) valueObj;
  var wrapperObj = cast (system::IRpcWrapperBase^) valueObj;
  wrapperObj = null;

  clientObj = null;
