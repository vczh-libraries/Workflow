module test;
using system::*;
using test::*;

var log = "";

func main():string
{
  var provider = new VariantProvider^();
  var subscription = bind(variant in provider.Value);
  subscription.Open();
  attach(subscription.ValueChanged, func(value:object):void
  {
    log = log & "[" & (value is int ? cast string (cast int value) : cast string value) & "]";
  });
  provider.Value = 1 of variant<int, string>;
  provider.Value = "text" of variant<int, string>;
  subscription.Close();
  provider.Value = 3 of variant<int, string>;
  return log;
}
