module Rpc;
using system::*;
using RpcWrapperTest::*;
using RpcCollection::Interface::Nested::PropDefault::*;

func serviceMain(lc : IRpcLifecycle*) : void
{
  var serviceObj = new (IService^)
  {
    var xsService : IValue^[][int] = {};

    func CheckServiceItems(xs : IValue^[][int]) : void
    {
      var ys = xs[0];
      for (i in range [0, ys.Count))
      {
        CheckValue(ys[i], i != 3, $"xsService[0][$(i)]");
      }
    }

    var _List : IValue^[][int] = null;

    override func GetServiceResult() : string
    {
      CheckServiceItems(xsService);
      return Print(xsService);
    }

    override func GetList() : IValue^[][int]
    {
      return _List;
    }

    override func SetList(xs : IValue^[][int]) : void
    {
      var xsObj : object = xs;
      if ((xsObj as (system::IRpcWrapperBase^) is null))
      {
        raise "Parameter xs should be a wrapper object in serviceMain";
      }

      xsService = xs;
      var ys = xs[0];
      ys.Add(MakeValue(4));
      _List = xs;
    }
  };
  lc.RegisterLocalService(rpc_GetIds()["RpcCollection::Interface::Nested::PropDefault::IService"], serviceObj);
}
