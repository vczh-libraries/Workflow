module Rpc;
using system::*;
using RpcWrapperTest::*;

func Print(xs : observe (observe int[])[]) : string
{
  var result = "";
  for (x in xs[0])
  {
    result = $"$(result)$(x)";
  }
  return result;
}

func clientMain(lc : IRpcLifecycle*) : string
{
  var xsOrigin : observe (observe int[])[] = {{1 2 3}};
  var service = cast (RpcCollectionOblist::Nested::Default::IService^) lc.RequestService("RpcCollectionOblist::Nested::Default::IService");
  var xs = service.DoList(xsOrigin);

  if (xs != xsOrigin)
  {
    raise "Return value xs should be the original parameter object in clientMain";
  }

  var xsClient = xs;
  var ys = xs[0];
  ys.Add(5);

  return
    $"[$(Print(xsOrigin))]" &
    $"[$(service.GetServiceResult())]" &
    $"[$(Print(xsClient))]";
}
