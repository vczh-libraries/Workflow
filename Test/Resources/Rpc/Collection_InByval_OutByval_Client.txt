module Rpc;
using system::*;
using RpcWrapperTest::*;

func Print(xs : int[]) : string
{
  var result = "";
  for (x in xs)
  {
    result = $"$(result)$(x)";
  }
  return result;
}

func clientMain(lc : IRpcLifecycle*) : string
{
  var xsOrigin = {1 2 3};
  var service = cast (RpcCollection::InByval::OutByval::IService^) lc.RequestService("RpcCollection::InByval::OutByval::IService");
  var xs = service.DoList(xsOrigin);
  var xsObj : object = xs;

  if (xs == xsOrigin)
  {
    raise "Return value xs should be a copied local object in clientMain";
  }

  if ((xsObj as (system::IRpcWrapperBase^) is not null))
  {
    raise "Return value xs should be a copied local object in clientMain";
  }

  var xsClient = xs;
  xs.Add(5);

  return
    $"[$(Print(xsOrigin))]" &
    $"[$(service.GetServiceResult())]" &
    $"[$(Print(xsClient))]";
}
