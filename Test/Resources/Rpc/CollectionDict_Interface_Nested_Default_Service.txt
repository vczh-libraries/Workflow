module Rpc;
using system::*;
using RpcWrapperTest::*;
using RpcCollectionDict::Interface::Nested::Default::*;

func serviceMain(lc : IRpcLifecycle*) : void
{
  var serviceObj = new (IService^)
  {
    var xsService : IValue^[int][int] = {};

    func CheckServiceItems(xs : IValue^[int][int]) : void
    {
      var ys = xs[0];
      for (i in range [1, ys.Count])
      {
        CheckValue(ys[i], i != 4, $"xsService[0][$(i)]");
      }
    }

    override func GetServiceResult() : string
    {
      CheckServiceItems(xsService);
      return Print(xsService);
    }

    override func DoList(xs : IValue^[int][int]) : IValue^[int][int]
    {
      var xsObj : object = xs;
      if ((xsObj as (system::IRpcWrapperBase^) is null))
      {
        raise "Parameter xs should be a wrapper object in serviceMain";
      }

      xsService = xs;
      var ys = xs[0];
      ys.Set(4, MakeValue("D"));
      return xs;
    }
  };
  lc.RegisterLocalService(rpc_GetIds()["RpcCollectionDict::Interface::Nested::Default::IService"], serviceObj);
}
