module Rpc;
using system::*;
using RpcWrapperTest::*;

func serviceMain(lc : IRpcLifecycle*) : void
{
  var serviceObj = new (RpcPropDefault::IService^)
  {
    var _Value : string = "A";

    override func GetValue() : string
    {
      return _Value;
    }

    override func SetValue(value : string) : void
    {
      _Value = value;
    }

    override func Signal() : void
    {
      ValueChanged();
    }
  };
  lc.RegisterLocalService(rpc_GetIds()["RpcPropDefault::IService"], serviceObj);
}

var s = "";

func ClientValueChanged() : void
{
  s = $"$(s)[ValueChanged]";
}

func clientMain(lc : IRpcLifecycle*) : string
{
  var service = cast (RpcPropDefault::IService^) lc.RequestService("RpcPropDefault::IService");
  attach(service.ValueChanged, ClientValueChanged);

  s = $"$(s)[$(service.Value)]";

  service.SetValue("B");
  s = $"$(s)[$(service.Value)]";

  service.Signal();
  s = $"$(s)[$(service.Value)]";

  return s;
}
