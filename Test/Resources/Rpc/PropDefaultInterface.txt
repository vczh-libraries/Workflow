module Rpc;
using system::*;
using RpcWrapperTest::*;
using RpcPropDefaultInterface::*;

namespace RpcPropDefaultInterface
{
  @rpc:Interface
  interface IValue
  {
    func GetValue() : string;
    @rpc:Dynamic
    prop Value : string {GetValue}
  }

  @rpc:Interface
  @rpc:Ctor
  interface IService
  {
    prop Value : IValue^ {GetValue : ValueChanged}
    func GetValue() : IValue^;

    func SetValue(value : IValue^) : void;
    func Signal() : void;
    event ValueChanged();
  }
}

func MakeValue(value : string) : IValue^
{
  return new (IValue^)
  {
    override func GetValue() : string
    {
      return value;
    }
  };
}

func CheckValue(value : IValue^, shouldBeWrapper : bool, name : string) : void
{
  var valueObj : object = value;
  if (shouldBeWrapper)
  {
    if ((valueObj as (system::IRpcWrapperBase^) is null))
    {
      raise $"$(name) should be a wrapper object";
    }
  }
  else
  {
    if ((valueObj as (system::IRpcWrapperBase^) is not null))
    {
      raise $"$(name) should be a local object";
    }
  }
}
