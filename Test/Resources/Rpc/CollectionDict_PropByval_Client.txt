module Rpc;
using system::*;
using RpcWrapperTest::*;

func Print(xs : string[int]) : string
{
  var result = "";
  for (key in range [1, xs.Count])
  {
    result = $"$(result)$(key)$(xs[key])";
  }
  return result;
}

func clientMain(lc : IRpcLifecycle*) : string
{
  var xsOrigin = {1:"A" 2:"B" 3:"C"};
  var service = cast (RpcCollectionDict::InByval::PropByval::IService^) lc.RequestService("RpcCollectionDict::InByval::PropByval::IService");
  service.List = xsOrigin;
  var xs = service.List;
  var xsObj : object = xs;

  if (xs == xsOrigin)
  {
    raise "Return value xs should be a copied local object in clientMain";
  }

  if ((xsObj as (system::IRpcWrapperBase^) is not null))
  {
    raise "Return value xs should be a copied local object in clientMain";
  }

  var xsClient = xs;
  xs.Set(5, "E");

  return
    $"[$(Print(xsOrigin))]" &
    $"[$(service.GetServiceResult())]" &
    $"[$(Print(xsClient))]";
}
