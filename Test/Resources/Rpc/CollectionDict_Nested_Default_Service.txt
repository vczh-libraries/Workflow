module Rpc;
using system::*;
using RpcWrapperTest::*;

func serviceMain(lc : IRpcLifecycle*) : void
{
  var serviceObj = new (RpcCollectionDict::Nested::Default::IService^)
  {
    var xsService : (string[int])[] = {};

    override func GetServiceResult() : string
    {
      return Print(xsService);
    }

    override func DoList(xs : (string[int])[]) : (string[int])[]
    {
      var xsObj : object = xs;
      if ((xsObj as (system::IRpcWrapperBase^) is not null))
      {
        raise "Parameter xs should be a copied local object in serviceMain";
      }

      xsService = xs;
      var ys = xs[0];
      ys.Set(4, "D");
      return xs;
    }
  };
  lc.RegisterLocalService(rpc_GetIds()["RpcCollectionDict::Nested::Default::IService"], serviceObj);
}
