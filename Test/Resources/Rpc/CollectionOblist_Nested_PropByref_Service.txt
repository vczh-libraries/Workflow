module Rpc;
using system::*;
using RpcWrapperTest::*;

func serviceMain(lc : IRpcLifecycle*) : void
{
  var serviceObj = new (RpcCollectionOblist::Nested::InByref::PropByref::IService^)
  {
    var xsService : observe (observe int[])[] = {};

    var _List : observe (observe int[])[] = null;

    override func GetServiceResult() : string
    {
      return Print(xsService);
    }

    override func GetList() : observe (observe int[])[]
    {
      return _List;
    }

    override func SetList(xs : observe (observe int[])[]) : void
    {
      var xsObj : object = xs;
      if ((xsObj as (system::IRpcWrapperBase^) is null))
      {
        raise "Parameter xs should be a wrapper object in serviceMain";
      }

      xsService = xs;
      var ys = xs[0];
      ys.Add(4);
      _List = xs;
    }
  };
  lc.RegisterLocalService(rpc_GetIds()["RpcCollectionOblist::Nested::InByref::PropByref::IService"], serviceObj);
}
