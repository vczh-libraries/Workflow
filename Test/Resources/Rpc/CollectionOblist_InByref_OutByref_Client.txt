module Rpc;
using system::*;
using RpcWrapperTest::*;
using RpcCollectionOblist::InByref::OutByref::*;

func Print5(xs : observe int[]) : string
{
  return $"$(xs[0])$(xs[1])$(xs[2])$(xs[3])$(xs[4])";
}

func clientMain(lc : IRpcLifecycle*) : string
{
  var xsOrigin : observe int[] = {1 2 3};
  var service = cast (IService^) lc.RequestService("RpcCollectionOblist::InByref::OutByref::IService");
  var xs = service.DoList(xsOrigin);

  if (xs != xsOrigin)
  {
    raise "Return value xs should be the original parameter object in clientMain";
  }

  var xsClient = xs;
  xs.Add(5);

  return
    $"[$(Print5(xsOrigin))]" &
    $"[$(service.GetServiceResult())]" &
    $"[$(Print5(xsClient))]";
}
