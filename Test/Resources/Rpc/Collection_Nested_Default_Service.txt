module Rpc;
using system::*;
using RpcWrapperTest::*;

func serviceMain(lc : IRpcLifecycle*) : void
{
  var serviceObj = new (RpcCollection::Nested::Default::IService^)
  {
    var xsService : (int[])[] = {};

    override func GetServiceResult() : string
    {
      return Print(xsService);
    }

    override func DoList(xs : (int[])[]) : (int[])[]
    {
      var xsObj : object = xs;
      if ((xsObj as (system::IRpcWrapperBase^) is not null))
      {
        raise "Parameter xs should be a copied local object in serviceMain";
      }

      xsService = xs;
      var ys = xs[0];
      ys.Add(4);
      return xs;
    }
  };
  lc.RegisterLocalService(rpc_GetIds()["RpcCollection::Nested::Default::IService"], serviceObj);
}
