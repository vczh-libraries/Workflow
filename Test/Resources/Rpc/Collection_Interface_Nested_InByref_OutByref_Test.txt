module Rpc;
using system::*;
using RpcWrapperTest::*;
using RpcCollection::Interface::Nested::InByref::OutByref::*;

func MakeValue(value : int) : IValue^
{
  return new (IValue^)
  {
    override func GetValue() : int
    {
      return value;
    }
  };
}

func CheckValue(value : IValue^, shouldBeWrapper : bool, name : string) : void
{
  var valueObj : object = value;
  if (shouldBeWrapper)
  {
    if ((valueObj as (system::IRpcWrapperBase^) is null))
    {
      raise $"$(name) should be a wrapper object";
    }
  }
  else
  {
    if ((valueObj as (system::IRpcWrapperBase^) is not null))
    {
      raise $"$(name) should be a local object";
    }
  }
}

func serviceMain(lc : IRpcLifecycle*) : void
{
  var serviceObj = new (IService^)
  {
    var xsService : IValue^[][int] = {};

    func CheckServiceItems(xs : IValue^[][int]) : void
    {
      var ys = xs[0];
      for (i in range [0, ys.Count))
      {
        CheckValue(ys[i], i != 3, $"xsService[0][$(i)]");
      }
    }

    override func GetServiceResult() : string
    {
      CheckServiceItems(xsService);
      return Print(xsService);
    }

    override func DoList(xs : IValue^[][int]) : IValue^[][int]
    {
      var xsObj : object = xs;
      if ((xsObj as (system::IRpcWrapperBase^) is null))
      {
        raise "Parameter xs should be a wrapper object in serviceMain";
      }

      xsService = xs;
      var ys = xs[0];
      ys.Add(MakeValue(4));
      return xs;
    }
  };
  lc.RegisterLocalService(rpc_GetIds()["RpcCollection::Interface::Nested::InByref::OutByref::IService"], serviceObj);
}

func CheckClientItems(xs : IValue^[][int]) : void
{
  var ys = xs[0];
  for (i in range [0, ys.Count))
  {
    CheckValue(ys[i], i == 3, $"xsClient[0][$(i)]");
  }
}

func Print(xs : IValue^[][int]) : string
{
  var result = "";
  for (x in xs[0])
  {
    result = $"$(result)$(x.Value)";
  }
  return result;
}

func clientMain(lc : IRpcLifecycle*) : string
{
  var ysOrigin : IValue^[] = {};
  ysOrigin.Add(MakeValue(1));
  ysOrigin.Add(MakeValue(2));
  ysOrigin.Add(MakeValue(3));
  var xsOrigin : IValue^[][int] = {};
  xsOrigin.Set(0, ysOrigin);
  var service = cast (IService^) lc.RequestService("RpcCollection::Interface::Nested::InByref::OutByref::IService");
  var xs = service.DoList(xsOrigin);

  if (xs != xsOrigin)
  {
    raise "Return value xs should be the original parameter object in clientMain";
  }

  var xsClient = xs;
  var ys = xs[0];
  ys.Add(MakeValue(5));
  CheckClientItems(xsClient);

  return
    $"[$(Print(xsOrigin))]" &
    $"[$(service.GetServiceResult())]" &
    $"[$(Print(xsClient))]";
}
