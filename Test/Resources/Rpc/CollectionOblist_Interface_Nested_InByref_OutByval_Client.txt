module Rpc;
using system::*;
using RpcWrapperTest::*;
using RpcCollectionOblist::Interface::Nested::InByref::OutByval::*;

func CheckClientItems(xs : (observe IValue^[])[int]) : void
{
  var ys = xs[0];
  for (i in range [0, ys.Count))
  {
    CheckValue(ys[i], i == 3, $"xsClient[0][$(i)]");
  }
}

func Print(xs : (observe IValue^[])[int]) : string
{
  var result = "";
  for (x in xs[0])
  {
    result = $"$(result)$(x.Value)";
  }
  return result;
}

func clientMain(lc : IRpcLifecycle*) : string
{
  var ysOrigin : observe IValue^[] = {};
  ysOrigin.Add(MakeValue(1));
  ysOrigin.Add(MakeValue(2));
  ysOrigin.Add(MakeValue(3));
  var xsOrigin : (observe IValue^[])[int] = {};
  xsOrigin.Set(0, ysOrigin);
  var service = cast (IService^) lc.RequestService("RpcCollectionOblist::Interface::Nested::InByref::OutByval::IService");
  var xs = service.DoList(xsOrigin);

  if (xs == xsOrigin)
  {
    raise "Return value xs should be a copied local object in clientMain";
  }

  var xsObj : object = xs;
  if ((xsObj as (system::IRpcWrapperBase^) is not null))
  {
    raise "Return value xs should not be a wrapper object in clientMain";
  }

  var xsClient = xs;
  var ys = xs[0];
  ys.Add(MakeValue(5));
  CheckClientItems(xsClient);

  return
    $"[$(Print(xsOrigin))]" &
    $"[$(service.GetServiceResult())]" &
    $"[$(Print(xsClient))]";
}
