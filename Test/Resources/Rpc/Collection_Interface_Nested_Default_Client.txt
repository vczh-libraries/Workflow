module Rpc;
using system::*;
using RpcWrapperTest::*;
using RpcCollection::Interface::Nested::Default::*;

func CheckClientItems(xs : IValue^[][int]) : void
{
  var ys = xs[0];
  for (i in range [0, ys.Count))
  {
    CheckValue(ys[i], i == 3, $"xsClient[0][$(i)]");
  }
}

func Print(xs : IValue^[][int]) : string
{
  var result = "";
  for (x in xs[0])
  {
    result = $"$(result)$(x.Value)";
  }
  return result;
}

func clientMain(lc : IRpcLifecycle*) : string
{
  var ysOrigin : IValue^[] = {};
  ysOrigin.Add(MakeValue(1));
  ysOrigin.Add(MakeValue(2));
  ysOrigin.Add(MakeValue(3));
  var xsOrigin : IValue^[][int] = {};
  xsOrigin.Set(0, ysOrigin);
  var service = cast (IService^) lc.RequestService("RpcCollection::Interface::Nested::Default::IService");
  var xs = service.DoList(xsOrigin);

  if (xs != xsOrigin)
  {
    raise "Return value xs should be the original parameter object in clientMain";
  }

  var xsClient = xs;
  var ys = xs[0];
  ys.Add(MakeValue(5));
  CheckClientItems(xsClient);

  return
    $"[$(Print(xsOrigin))]" &
    $"[$(service.GetServiceResult())]" &
    $"[$(Print(xsClient))]";
}
