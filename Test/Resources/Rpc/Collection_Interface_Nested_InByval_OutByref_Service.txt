module Rpc;
using system::*;
using RpcWrapperTest::*;
using RpcCollection::Interface::Nested::InByval::OutByref::*;

func serviceMain(lc : IRpcLifecycle*) : void
{
  var serviceObj = new (IService^)
  {
    var xsService : IValue^[][int] = {};

    func CheckServiceItems(xs : IValue^[][int]) : void
    {
      var ys = xs[0];
      for (i in range [0, ys.Count))
      {
        CheckValue(ys[i], i != 3, $"xsService[0][$(i)]");
      }
    }

    override func GetServiceResult() : string
    {
      CheckServiceItems(xsService);
      return Print(xsService);
    }

    override func DoList(xs : IValue^[][int]) : IValue^[][int]
    {
      var xsObj : object = xs;
      if ((xsObj as (system::IRpcWrapperBase^) is not null))
      {
        raise "Parameter xs should be a copied local object in serviceMain";
      }

      xsService = xs;
      var ys = xs[0];
      ys.Add(MakeValue(4));
      return xs;
    }
  };
  lc.RegisterLocalService(rpc_GetIds()["RpcCollection::Interface::Nested::InByval::OutByref::IService"], serviceObj);
}
