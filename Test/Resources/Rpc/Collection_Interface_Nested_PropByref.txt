module Rpc;
using system::*;
using RpcWrapperTest::*;
using RpcCollection::Interface::Nested::InByref::PropByref::*;

namespace RpcCollection
{
  namespace Interface
  {
    namespace Nested
    {
      namespace InByref
      {
        namespace PropByref
        {
          @rpc:Interface
          interface IValue
          {
            func GetValue() : int;
            @rpc:Dynamic
            prop Value : int {GetValue}
          }

          @rpc:Interface
          @rpc:Ctor
          interface IService
          {
            func GetServiceResult() : string;
            @rpc:Byref
            @rpc:Dynamic
            prop List : IValue^[][int] {not observe}
          }
        }
      }
    }
  }
}

func MakeValue(value : int) : IValue^
{
  return new (IValue^)
  {
    override func GetValue() : int
    {
      return value;
    }
  };
}

func CheckValue(value : IValue^, shouldBeWrapper : bool, name : string) : void
{
  var valueObj : object = value;
  if (shouldBeWrapper)
  {
    if ((valueObj as (system::IRpcWrapperBase^) is null))
    {
      raise $"$(name) should be a wrapper object";
    }
  }
  else
  {
    if ((valueObj as (system::IRpcWrapperBase^) is not null))
    {
      raise $"$(name) should be a local object";
    }
  }
}
