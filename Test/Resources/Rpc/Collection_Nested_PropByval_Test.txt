module Rpc;
using system::*;
using RpcWrapperTest::*;

func serviceMain(lc : IRpcLifecycle*) : void
{
  var serviceObj = new (RpcCollection::Nested::InByval::PropByval::IService^)
  {
    var xsService : (int[])[] = {};

    var _List : (int[])[] = null;

    override func GetServiceResult() : string
    {
      return Print(xsService);
    }

    override func GetList() : (int[])[]
    {
      return _List;
    }

    override func SetList(xs : (int[])[]) : void
    {
      var xsObj : object = xs;
      if ((xsObj as (system::IRpcWrapperBase^) is not null))
      {
        raise "Parameter xs should be a copied local object in serviceMain";
      }

      xsService = xs;
      var ys = xs[0];
      ys.Add(4);
      _List = xs;
    }
  };
  lc.RegisterLocalService(rpc_GetIds()["RpcCollection::Nested::InByval::PropByval::IService"], serviceObj);
}

func Print(xs : (int[])[]) : string
{
  var result = "";
  for (x in xs[0])
  {
    result = $"$(result)$(x)";
  }
  return result;
}

func clientMain(lc : IRpcLifecycle*) : string
{
  var xsOrigin = {{1 2 3}};
  var service = cast (RpcCollection::Nested::InByval::PropByval::IService^) lc.RequestService("RpcCollection::Nested::InByval::PropByval::IService");
  service.List = xsOrigin;
  var xs = service.List;

  if (xs == xsOrigin)
  {
    raise "Return value xs should be a copied local object in clientMain";
  }

  var xsObj : object = xs;

  if ((xsObj as (system::IRpcWrapperBase^) is not null))
  {
    raise "Return value xs should be a copied local object in clientMain";
  }

  var xsClient = xs;
  var ys = xs[0];
  ys.Add(5);

  return
    $"[$(Print(xsOrigin))]" &
    $"[$(service.GetServiceResult())]" &
    $"[$(Print(xsClient))]";
}
