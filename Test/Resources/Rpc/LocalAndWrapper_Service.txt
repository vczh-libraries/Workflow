module Rpc;
using system::*;
using RpcWrapperTest::*;

func serviceMain(lc : IRpcLifecycle*) : void
{
  var service = new (RpcWrapperTest::IService^)
  {
    var _ServiceObj1 : IObj1^ = new IObj1^{};
    var _ServiceObj2 : IObj2^ = new IObj2^{};
    var _ServiceReceived_Obj1 : IObj1^ = null;
    var _ServiceReceived_Obj2 : IObj2^ = null;

    override func GetServiceResult() : string
    {
      return
        $"[$(_ServiceReceived_Obj1 == _ServiceObj1)]" &
        $"[$(_ServiceReceived_Obj2 == _ServiceObj2)]";
    }

    override func Exchange1(o : IObj1^) : IObj2^
    {
      var wrapperObj : object = o;
      if ((wrapperObj as (system::IRpcWrapperBase^) is null))
      {
        raise "IObj1(o) should be a wrapper object in serviceMain";
      }
      _ServiceReceived_Obj1 = o;
      return _ServiceObj2;
    }

    override func Exchange2(o : IObj2^) : IObj1^
    {
      var wrapperObj : object = o;
      if ((wrapperObj as (system::IRpcWrapperBase^) is not null))
      {
        raise "IObj2(o) should be a local object in serviceMain";
      }
      _ServiceReceived_Obj2 = o;
      return _ServiceReceived_Obj1;
    }
  };
  lc.RegisterLocalService(rpc_GetIds()["RpcWrapperTest::IService"], service);
}
