module Rpc;
using system::*;
using RpcFailDoubleRegistrationTest::*;

func serviceMain(lc : IRpcLifecycle*) : void
{
  var typeId = rpc_GetIds()["RpcFailDoubleRegistrationTest::IService"];
  var serviceObj = new (RpcFailDoubleRegistrationTest::IService^)
  {
    var serviceResult : string = "";

    override func GetResult() : string
    {
      if (serviceResult == "")
      {
        try
        {
          lc.RegisterLocalService(typeId, cast (RpcFailDoubleRegistrationTest::IService^) this);
          serviceResult = "[no exception]";
        }
        catch (ex)
        {
          serviceResult = "[exception]";
        }
      }
      return serviceResult;
    }
  };
  lc.RegisterLocalService(typeId, serviceObj);
  serviceObj.GetResult();
}

func clientMain(lc : IRpcLifecycle*) : string
{
  var service = cast (RpcFailDoubleRegistrationTest::IService^) lc.RequestService("RpcFailDoubleRegistrationTest::IService");
  return service.GetResult();
}
