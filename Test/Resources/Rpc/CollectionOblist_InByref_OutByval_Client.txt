module Rpc;
using system::*;
using RpcWrapperTest::*;
using RpcCollectionOblist::InByref::OutByval::*;

func Print4(xs : observe int[]) : string
{
  return $"$(xs[0])$(xs[1])$(xs[2])$(xs[3])";
}

func Print5(xs : observe int[]) : string
{
  return $"$(xs[0])$(xs[1])$(xs[2])$(xs[3])$(xs[4])";
}

func clientMain(lc : IRpcLifecycle*) : string
{
  var xsOrigin : observe int[] = {1 2 3};
  var service = cast (IService^) lc.RequestService("RpcCollectionOblist::InByref::OutByval::IService");
  var xs = service.DoList(xsOrigin);

  if (xs == xsOrigin)
  {
    raise "Return value xs should be a copied local object in clientMain";
  }

  var xsObj : object = xs;
  if ((xsObj as (system::IRpcWrapperBase^) is not null))
  {
    raise "Return value xs should not be a wrapper object in clientMain";
  }

  var xsClient = xs;
  xs.Add(5);

  return
    $"[$(Print4(xsOrigin))]" &
    $"[$(service.GetServiceResult())]" &
    $"[$(Print5(xsClient))]";
}
